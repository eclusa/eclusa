**Specific changes:**

Add a new function `read_workspace_files(workspace_dir: str) -> list[tuple[str, str]]`:
```python
_SKIP_DIRS = {".git", "__pycache__", "node_modules", ".venv", "venv", ".pytest_cache"}
_SKIP_EXTENSIONS = {".pyc", ".pyo", ".egg-info"}

def read_workspace_files(workspace_dir: str) -> list[tuple[str, str]]:
    """Read all files from workspace directory, returning (relative_path, content) pairs."""
    workspace = Path(workspace_dir)
    files: list[tuple[str, str]] = []
    for path in sorted(workspace.rglob("*")):
        if not path.is_file():
            continue
        # Skip hidden/build dirs
        if any(part in _SKIP_DIRS for part in path.parts):
            continue
        if path.suffix in _SKIP_EXTENSIONS:
            continue
        rel_path = str(path.relative_to(workspace))
        try:
            content = path.read_text(encoding="utf-8")
            files.append((rel_path, content))
        except (UnicodeDecodeError, PermissionError):
            continue  # Skip binary files
    return files
```
