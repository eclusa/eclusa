```python
"""executor/scc.py — SCC cascade template creation (SCC-01, D-01, D-02).

create_scc_cascade(intent_id, actor_id, conn): creates a 7-stage cascade.
7 stages: refine → intent_validation_fanout → match → cohere → formalize → derive → generate.
All stages are type='narrowing', state='pending'.
Each stage input JSONB has 'scc_stage' key (D-02) plus stage-specific keys.
The fanout stage input includes 'refine_stage_id' for the dispatch handler to read (Pitfall 2).
"""
import json
import uuid
import asyncpg

async def create_scc_cascade(intent_id: str, actor_id: str, conn: asyncpg.Connection) -> str:
    cascade_id = str(uuid.uuid4())

    async with conn.transaction():
        await conn.execute("""
            INSERT INTO cascade (id, intent_id, shape, state)
            VALUES ($1::uuid, $2::uuid, '{}', 'active')
        """, cascade_id, intent_id)

        # Stage 1: Refine — no depends_on
        refine_row = await conn.fetchrow("""
            INSERT INTO stage (id, cascade_id, type, state, depends_on, input)
            VALUES (gen_random_uuid(), $1::uuid, 'narrowing', 'pending', '{}', $2::jsonb)
            RETURNING id
        """, cascade_id, json.dumps({"scc_stage": "refine"}))
        refine_id = str(refine_row["id"])

        # Fanout stage — depends on refine, includes refine_stage_id for handler
        fanout_row = await conn.fetchrow("""
            INSERT INTO stage (id, cascade_id, type, state, depends_on, input)
            VALUES (gen_random_uuid(), $1::uuid, 'narrowing', 'pending', ARRAY[$2::uuid], $3::jsonb)
            RETURNING id
        """, cascade_id, refine_id, json.dumps({
            "scc_stage": "intent_validation_fanout",
            "refine_stage_id": refine_id,
        }))
        fanout_id = str(fanout_row["id"])

        # Stage 2: Match — depends on fanout
        match_row = await conn.fetchrow("""
            INSERT INTO stage (id, cascade_id, type, state, depends_on, input)
            VALUES (gen_random_uuid(), $1::uuid, 'narrowing', 'pending', ARRAY[$2::uuid], $3::jsonb)
            RETURNING id
        """, cascade_id, fanout_id, json.dumps({"scc_stage": "match"}))
        match_id = str(match_row["id"])

        # Stage 3: Cohere — depends on match
        cohere_row = await conn.fetchrow("""
            INSERT INTO stage (id, cascade_id, type, state, depends_on, input)
            VALUES (gen_random_uuid(), $1::uuid, 'narrowing', 'pending', ARRAY[$2::uuid], $3::jsonb)
            RETURNING id
        """, cascade_id, match_id, json.dumps({"scc_stage": "cohere"}))
        cohere_id = str(cohere_row["id"])

        # Stage 4: Formalize — depends on cohere
        formalize_row = await conn.fetchrow("""
            INSERT INTO stage (id, cascade_id, type, state, depends_on, input)
            VALUES (gen_random_uuid(), $1::uuid, 'narrowing', 'pending', ARRAY[$2::uuid], $3::jsonb)
            RETURNING id
        """, cascade_id, cohere_id, json.dumps({"scc_stage": "formalize"}))
        formalize_id = str(formalize_row["id"])

        # Stage 5: Derive — depends on formalize
        derive_row = await conn.fetchrow("""
            INSERT INTO stage (id, cascade_id, type, state, depends_on, input)
            VALUES (gen_random_uuid(), $1::uuid, 'narrowing', 'pending', ARRAY[$2::uuid], $3::jsonb)
            RETURNING id
        """, cascade_id, formalize_id, json.dumps({"scc_stage": "derive"}))
        derive_id = str(derive_row["id"])

        # Stage 6: Generate — depends on derive
        await conn.execute("""
            INSERT INTO stage (id, cascade_id, type, state, depends_on, input)
            VALUES (gen_random_uuid(), $1::uuid, 'narrowing', 'pending', ARRAY[$2::uuid], $3::jsonb)
        """, cascade_id, derive_id, json.dumps({"scc_stage": "generate"}))

    return cascade_id
```
