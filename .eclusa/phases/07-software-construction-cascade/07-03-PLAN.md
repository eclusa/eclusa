```python
"""harness/formalize.py — GHC async subprocess for Haskell constraint verification.

verify_constraints_with_ghc: writes haskell_source to /workspace tempfile,
  calls `docker exec ghc-sidecar ghc -fno-code` via asyncio.create_subprocess_exec,
  returns (success, output). Non-blocking (D-13). Pitfall 3: normalizes paths.

run_formalize_with_ghc: drives the LLM → GHC retry loop (up to GHC_MAX_RETRIES=5, D-14).
  Returns (success, final_haskell_source, final_output).

_GHC_SEMAPHORE: asyncio.Semaphore(3) prevents concurrent OOM (Pitfall 5).
"""
import asyncio
import logging
import os
import tempfile
from pathlib import Path
from typing import Any

from pydantic_ai import Agent

logger = logging.getLogger(__name__)

GHC_CONTAINER_NAME = "ghc-sidecar"
GHC_WORKSPACE = "/workspace"
GHC_TIMEOUT_SECONDS = 10
GHC_MAX_RETRIES = 5

_GHC_SEMAPHORE = asyncio.Semaphore(3)


class FormalizeError(Exception):
    """Raised when GHC rejects constraints after GHC_MAX_RETRIES attempts."""


async def verify_constraints_with_ghc(haskell_source: str) -> tuple[bool, str]:
    """Write haskell_source to /workspace tempfile, call ghc -fno-code via docker exec.

    Returns (success, output). output = stdout + stderr combined.
    Path in output normalized to 'constraints.hs' (Pitfall 3).
    Uses asyncio.create_subprocess_exec — non-blocking (D-13).
    Semaphore limits concurrent GHC calls to 3 (Pitfall 5).
    """
    async with _GHC_SEMAPHORE:
        fd, local_path = tempfile.mkstemp(suffix=".hs", dir=GHC_WORKSPACE)
        try:
            with os.fdopen(fd, "w") as f:
                f.write(haskell_source)

            container_path = local_path  # same path — shared volume
            basename = os.path.basename(local_path)

            proc = await asyncio.create_subprocess_exec(
                "docker", "exec", GHC_CONTAINER_NAME,
                "ghc", "-fno-code", container_path,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
            )
            try:
                stdout, stderr = await asyncio.wait_for(
                    proc.communicate(), timeout=GHC_TIMEOUT_SECONDS
                )
            except asyncio.TimeoutError:
                proc.kill()
                return False, "GHC compilation timed out"

            output = (stdout + stderr).decode("utf-8", errors="replace")
            # Normalize tempfile path in error messages (Pitfall 3)
            output = output.replace(local_path, "constraints.hs")
            output = output.replace(basename, "constraints.hs")
            return proc.returncode == 0, output

        finally:
            Path(local_path).unlink(missing_ok=True)


SYSTEM_PROMPT = (
    "You are a Haskell type system expert. "
    "Translate business constraints into Haskell type-level proofs using -fno-code. "
    "Use newtype wrappers for domain types. "
    "Constraint functions must type-check under ghc -fno-code with no imports beyond Prelude. "
    "Output only valid Haskell source code with no markdown fencing."
)

INITIAL_PROMPT_TEMPLATE = (
    "Domain context:\n{context}\n\n"
    "Translate the above constraints into Haskell types and constraint functions. "
    "Ensure every constraint can be verified by GHC type-checker."
)

RETRY_PROMPT_TEMPLATE = (
    "The previous Haskell had type errors. Fix them.\n\n"
    "GHC errors:\n{errors}\n\n"
    "Previous Haskell:\n{previous}\n\n"
    "Output corrected Haskell only."
)


async def run_formalize_with_ghc(
    context: str,
    model: str = "anthropic:claude-3-5-haiku-latest",
) -> tuple[bool, str, str]:
    """Drive LLM → GHC retry loop until constraints compile or max retries exceeded.

    Returns (success, haskell_source, ghc_output).
    Raises FormalizeError after GHC_MAX_RETRIES consecutive failures.
    """
    agent = Agent(model, system_prompt=SYSTEM_PROMPT)

    # First attempt: draft from context
    result = await agent.run(INITIAL_PROMPT_TEMPLATE.format(context=context))
    haskell_source = str(result.output).strip()

    for attempt in range(GHC_MAX_RETRIES):
        success, output = await verify_constraints_with_ghc(haskell_source)
        if success:
            logger.info("GHC accepted constraints on attempt %d", attempt + 1)
            return True, haskell_source, output

        logger.warning("GHC rejected constraints (attempt %d/%d): %s", attempt + 1, GHC_MAX_RETRIES, output[:200])

        if attempt + 1 >= GHC_MAX_RETRIES:
            break

        # Retry: pass verbatim GHC errors back to LLM (D-14)
        result = await agent.run(
            RETRY_PROMPT_TEMPLATE.format(errors=output, previous=haskell_source)
        )
        haskell_source = str(result.output).strip()

    raise FormalizeError(
        f"GHC rejected constraints after {GHC_MAX_RETRIES} attempts. Last error:\n{output}"
    )
```
