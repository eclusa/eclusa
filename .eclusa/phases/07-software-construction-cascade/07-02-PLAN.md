```python
from executor.scc_handlers import (
    dispatch_scc_refine, dispatch_scc_match, dispatch_scc_cohere,
    dispatch_scc_formalize, dispatch_scc_derive, dispatch_scc_generate,
    dispatch_scc_fanout,
)

async def dispatch_narrowing(conn: asyncpg.Connection, stage: dict, actor_id: str) -> None:
    stage_input = stage.get("input") or {}
    if isinstance(stage_input, str):
        import json
        stage_input = json.loads(stage_input)
    scc_stage = stage_input.get("scc_stage")

    if scc_stage == "refine":
        await dispatch_scc_refine(conn, stage, actor_id)
    elif scc_stage == "intent_validation_fanout":
        await dispatch_scc_fanout(conn, stage, actor_id)
    elif scc_stage == "match":
        await dispatch_scc_match(conn, stage, actor_id)
    elif scc_stage == "cohere":
        await dispatch_scc_cohere(conn, stage, actor_id)
    elif scc_stage == "formalize":
        await dispatch_scc_formalize(conn, stage, actor_id)
    elif scc_stage == "derive":
        await dispatch_scc_derive(conn, stage, actor_id)
    elif scc_stage == "generate":
        await dispatch_scc_generate(conn, stage, actor_id)
    else:
        # Legacy non-SCC narrowing stages: resolve immediately (backward compat, Pitfall 4)
        await _resolve_stage_immediately(conn, stage, actor_id)

async def _resolve_stage_immediately(conn, stage, actor_id):
    """Backward-compatible immediate resolve for non-SCC stages."""
    async with conn.transaction():
        await conn.execute("""
            UPDATE stage SET state = 'resolved', resolved_at = NOW(), resolved_by = $1::uuid
            WHERE id = $2::uuid
        """, actor_id, stage["id"])
        await conn.execute("""
            INSERT INTO ledger_entry (id, stage_id, cascade_id, actor_id, type, content, schema_version)
            SELECT gen_random_uuid(), s.id, s.cascade_id, $1::uuid,
                   'stage_state_changed',
                   jsonb_build_object('old_state', 'active', 'new_state', 'resolved', 'note', 'legacy stub'),
                   $2
            FROM stage s WHERE s.id = $3::uuid
        """, actor_id, SYSTEM_SCHEMA_VERSION, stage["id"])
```

**executor/scc_handlers.py** — create new file with:

dispatch_scc_refine(conn, stage, actor_id):
- Reads stage.cascade_id and looks up intent_id via `SELECT intent_id FROM cascade WHERE id = $1`
- Reads raw_intent from stage.input.get("raw_intent", "") as the initial user message
- Calls start_work_session(conn, stage_id, intent_id, cascade_id, model="anthropic:claude-3-5-haiku-latest", actor_id)
- Creates pydantic-ai Agent with model, system prompt: "You are a requirements analyst. Narrow the user's noisy intent into a structured scope document with: goals, non-goals, constraints, open questions."
- Calls run_session_turn(conn, session_id, agent, raw_intent or "Refine this intent into a scope document.")
- Calls complete_work_session(conn, session_id, actor_id)
- Writes scope_doc to stage.output: UPDATE stage SET output = jsonb_build_object('scope_doc', $1, 'session_id', $2) WHERE id = $3
- Resolves stage: UPDATE stage SET state='resolved', resolved_at=NOW(), resolved_by=$actor_id
- Writes ledger entry stage_state_changed

dispatch_scc_match(conn, stage, actor_id):
- Reads query_text from stage.input.get("query_text", "") — the domain concept to search
- Generates embedding via embed_texts([query_text]) from schema_commons.embed — import inline
- Calls search_schema_commons(query_text, query_embedding, conn, limit=10)
- Serializes results to list of dicts: [r.model_dump() for r in results]
- Writes to stage.output: UPDATE stage SET output = $1::jsonb WHERE id = $2
- Resolves stage + ledger entry

dispatch_scc_cohere(conn, stage, actor_id):
- Reads matched_sources from stage.input.get("matched_sources", []) — passed from prior Match stage output
- Prepares context: f"Matched schema sources:\n{json.dumps(matched_sources, indent=2)}"
- context_hash via hash_context() from judgment.pass_
- Calls run_judgment_pass(model="anthropic:claude-opus-4-5", prepared_context, prompt="Identify composition issues: type boundary conflicts, auth model mismatches, data friction points between these matched sources.")
- Calls create_judgment_pass_record(conn, stage_id, model, context_ref=f"inline:{context_hash}", context_hash, prompt, verdict, actor_id)
- Writes to stage.output: UPDATE stage SET output = $1::jsonb WHERE id = $2 with verdict.model_dump()
- Resolves stage + ledger entry

All three handlers write a ledger entry type='stage_state_changed' with old_state='active', new_state='resolved' after updating stage state. Use SYSTEM_SCHEMA_VERSION = "0002".

Import pattern for embed_texts (used in Match):
```python
from schema_commons.embed import embed_texts
```
