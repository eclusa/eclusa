    ```python
    @router.get("/{cascade_id}/stages/{stage_id}/trace", response_model=TraceChain)
    async def get_stage_trace(cascade_id: UUID, stage_id: UUID, request: Request) -> TraceChain:
        pool = request.app.state.pool
        # Fetch stage → cascade → intent chain
        row = await pool.fetchrow(
            """
            SELECT
                s.id AS stage_id,
                s.type AS stage_type,
                s.cascade_id,
                c.intent_id,
                i.raw AS intent_raw
            FROM stage s
            JOIN cascade c ON c.id = s.cascade_id
            JOIN intent i ON i.id = c.intent_id
            WHERE s.id = $1::uuid AND s.cascade_id = $2::uuid
            """,
            stage_id, cascade_id,
        )
        if row is None:
            raise HTTPException(status_code=404, detail="Stage not found")

        # Fetch work_session linked to this stage
        session_row = await pool.fetchrow(
            """
            SELECT id FROM work_session
            WHERE $1::uuid = ANY(stage_ids)
            LIMIT 1
            """,
            stage_id,
        )

        # Fetch one artifact linked to the session (most recent)
        artifact_row = None
        if session_row is not None:
            artifact_row = await pool.fetchrow(
                """
                SELECT id, payload->>'type' AS artifact_type
                FROM artifact
                WHERE session_id = $1::uuid
                ORDER BY created_at DESC, id DESC
                LIMIT 1
                """,
                session_row["id"],
            )

        hops = []
        if artifact_row is not None:
            hops.append(TraceHop(
                type="artifact",
                id=str(artifact_row["id"]),
                label=f"artifact: {artifact_row['artifact_type'] or 'output'}",
                href=None,
            ))
        if session_row is not None:
            sid = str(session_row["id"])
            hops.append(TraceHop(
                type="session",
                id=sid,
                label=f"session: {sid[:8]}",
                href=f"/sessions/{sid}",
            ))
        stage_id_str = str(row["stage_id"])
        hops.append(TraceHop(
            type="stage",
            id=stage_id_str,
            label=f"stage: {stage_id_str[:8]}",
            href=None,
        ))
        cascade_id_str = str(row["cascade_id"])
        hops.append(TraceHop(
            type="cascade",
            id=cascade_id_str,
            label=f"cascade: {cascade_id_str[:8]}",
            href=f"/cascades/{cascade_id_str}",
        ))
        intent_raw = row["intent_raw"] or ""
        intent_label = intent_raw[:40] + "..." if len(intent_raw) > 40 else intent_raw
        hops.append(TraceHop(
            type="intent",
            id=str(row["intent_id"]),
            label=f"intent: {intent_label}",
            href=None,
        ))

        return TraceChain(stage_id=stage_id_str, hops=hops)
    ```

    **B. Add /ws/stages/{stage_id} to adapters/web/api/ws.py**

    Add a new WebSocket endpoint after the existing `stream_session_messages` route. This endpoint:
    - Accepts JWT via `?token=` (same pattern as existing endpoint)
    - Looks up the work_session linked to the stage via `WHERE $1::uuid = ANY(stage_ids)`
    - If no session found, sends `{"type": "no_session"}` and closes
    - Streams artifacts from that session using the same `_fetch_messages` helper
    - Sends `{"type": "chunk", "content": message_content_text}` for each new artifact where role = 'assistant'
    - Sends `{"type": "done"}` when stage state transitions from 'active' to anything else (poll stage state every 2s alongside messages)
    - Handles WebSocketDisconnect gracefully

    ```python
    @router.websocket("/ws/stages/{stage_id}")
    async def stream_stage_output(websocket: WebSocket, stage_id: UUID) -> None:
        await websocket.accept()

        token = websocket.query_params.get("token")
        if not token:
            await websocket.send_json({"error": "unauthorized"})
            await websocket.close(code=1008)
            return
        try:
            jwt.decode(token, get_jwt_secret(), algorithms=[JWT_ALGORITHM])
        except jwt.InvalidTokenError:
            await websocket.send_json({"error": "unauthorized"})
            await websocket.close(code=1008)
            return

        pool = websocket.app.state.pool
        try:
            async with pool.acquire() as conn:
                session_row = await conn.fetchrow(
                    "SELECT id FROM work_session WHERE $1::uuid = ANY(stage_ids) LIMIT 1",
                    stage_id,
                )
            if session_row is None:
                await websocket.send_json({"type": "no_session"})
                await websocket.close()
                return

            session_id = session_row["id"]
            last_created_at = datetime(1970, 1, 1, tzinfo=timezone.utc)
            last_id = UUID(int=0)

            while True:
                await asyncio.sleep(2)
                async with pool.acquire() as conn:
                    new_rows = await _fetch_messages(conn, session_id, last_created_at, last_id)
                    stage_row = await conn.fetchrow(
                        "SELECT state FROM stage WHERE id = $1::uuid", stage_id
                    )

                for row in new_rows:
                    content = row["content"]
                    if isinstance(content, dict):
                        text = content.get("text") or content.get("content") or str(content)
                    else:
                        text = str(content)
                    await websocket.send_json({"type": "chunk", "content": text, "role": row["role"]})
                    last_created_at = row["created_at"]
                    last_id = row["id"]

                if stage_row and stage_row["state"] not in ("active", "pending"):
                    await websocket.send_json({"type": "done", "state": stage_row["state"]})
                    break
        except WebSocketDisconnect:
            return
    ```

    **A. Create ui/src/api/trace.ts**:

    ```typescript
    import { useQuery } from '@tanstack/react-query'
    import { useAuth } from '@/auth/AuthProvider'

    export interface TraceHop {
      type: 'artifact' | 'session' | 'stage' | 'cascade' | 'intent'
      id: string
      label: string
      href: string | null
    }

    export interface TraceChain {
      stage_id: string
      hops: TraceHop[]
    }

    async function requestJson<T>(url: string, headers: Record<string, string>) {
      const response = await fetch(url, { headers })
      if (!response.ok) throw new Error(`Request failed (${response.status})`)
      return response.json() as Promise<T>
    }

    export function fetchTraceChain(cascadeId: string, stageId: string, headers: Record<string, string>) {
      return requestJson<TraceChain>(`/api/cascades/${cascadeId}/stages/${stageId}/trace`, headers)
    }

    export function useTraceChain(cascadeId: string | null, stageId: string | null) {
      const { getAuthHeader } = useAuth()
      return useQuery<TraceChain>({
        queryKey: ['trace', cascadeId, stageId],
        queryFn: () => {
          if (!cascadeId || !stageId) throw new Error('ids required')
          return fetchTraceChain(cascadeId, stageId, getAuthHeader())
        },
        enabled: Boolean(cascadeId) && Boolean(stageId),
      })
    }
    ```

      ```typescript
      useEffect(() => {
        if (stage.state !== 'active') return
        const token = localStorage.getItem('eclusa_token') ?? ''
        const protocol = location.protocol === 'https:' ? 'wss:' : 'ws:'
        const ws = new WebSocket(`${protocol}//${location.host}/ws/stages/${stage.id}?token=${token}`)
        setWsConnected(true)
        ws.onmessage = (event) => {
          const msg = JSON.parse(event.data) as { type: string; content?: string; state?: string }
          if (msg.type === 'chunk' && msg.content) {
            setStreamedContent(prev => prev + msg.content)
          }
          if (msg.type === 'done') {
            ws.close()
            setWsConnected(false)
          }
        }
        ws.onclose = () => setWsConnected(false)
        return () => ws.close()
      }, [stage.id, stage.state])
      ```
