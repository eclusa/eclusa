"""
judgment/context_prep.py — Local context preparation for judgment passes.

No model call. Strips tool noise, summarizes long middles, foregrounds
decisions. Runs on executor. JUDG-03, D-13, D-14.
"""
import json
import logging

logger = logging.getLogger(__name__)


def prepare_context(message_history: list[dict], max_chars: int = 8000) -> str:
    """Prepare message history for judgment pass consumption.

    Strips tool call noise (kind='tool-call', kind='tool-return').
    Keeps user requests and model text responses.
    Truncates long middles: keep first 20% + last 60% if over max_chars.

    Returns a plain text string — the judgment prompt prefix.
    """
    # Extract readable messages only
    readable = []
    for msg in message_history:
        kind = msg.get("kind", "")
        if kind in ("tool-call", "tool-return"):
            continue  # strip noise
        role = "User" if kind == "request" else "Assistant"
        # Extract text content — may be list of parts or plain string
        content = msg.get("content", "")
        if isinstance(content, list):
            # pydantic-ai platform format: content is list of part dicts
            parts = [p.get("content", "") for p in content if isinstance(p, dict) and "content" in p]
            text = " ".join(str(p) for p in parts if p)
        else:
            text = str(content)
        if text.strip():
            readable.append(f"{role}: {text.strip()}")

    full_text = "\n\n".join(readable)

    if len(full_text) <= max_chars:
        return full_text

    # Truncate: keep first 20% and last 60%
    keep_start = int(max_chars * 0.20)
    keep_end = int(max_chars * 0.60)
    truncated = (
        full_text[:keep_start]
        + f"\n\n[... {len(full_text) - keep_start - keep_end} chars omitted ...]\n\n"
        + full_text[-keep_end:]
    )
    logger.debug("Context truncated from %d to %d chars", len(full_text), len(truncated))
    return truncated
```

# tests/test_judgment_pass.py
import pytest
from pydantic_ai.models.test import TestModel
from pydantic_ai import Agent
from judgment.pass_ import VerdictModel, hash_context, run_judgment_pass
from judgment.context_prep import prepare_context

@pytest.mark.asyncio
async def test_judgment_returns_structured_verdict():
    # TestModel returns a fixed response; output_type enforces schema
    model = TestModel()
    agent = Agent(model, output_type=VerdictModel)
    # TestModel needs to know what to return — configure via custom_output_text
    # or use Agent.run() and catch the schema enforcement
    # pydantic-ai TestModel auto-generates valid output for the output_type schema
    result = await agent.run("evaluate this context")
    verdict = result.output
    assert isinstance(verdict, VerdictModel)
    assert verdict.decision in ("approve", "reject", "needs_clarification")
    assert 0.0 <= verdict.confidence <= 1.0
    assert isinstance(verdict.conditions, list)

@pytest.mark.asyncio
async def test_judgment_uses_prepared_context():
    history = [
        {"kind": "request", "content": "analyze security"},
        {"kind": "response", "content": "I found 3 issues"},
    ]
    ctx = prepare_context(history)
    assert "analyze security" in ctx
    assert "I found 3 issues" in ctx
    # Tool noise stripped
    history_with_noise = history + [{"kind": "tool-call", "content": "exec_bash"}]
    ctx2 = prepare_context(history_with_noise)
    assert "exec_bash" not in ctx2

def test_context_hash_computed():
    h1 = hash_context("hello world")
    h2 = hash_context("hello world")
    h3 = hash_context("different content")
    assert h1 == h2          # deterministic
    assert h1 != h3          # different inputs differ
    assert len(h1) >= 32     # at least sha256-length hex string

@pytest.mark.asyncio
async def test_judgment_has_no_write_tools():
    model = TestModel()
    agent = Agent(model, output_type=VerdictModel)
    # Verify no tools registered on the agent
    # pydantic-ai exposes _tools or similar — check agent has no tools
    tool_names = list(agent._function_tools.keys()) if hasattr(agent, "_function_tools") else []
    assert tool_names == [], f"Judgment agent must have no tools, found: {tool_names}"
```

Note on TestModel behavior: pydantic-ai's TestModel generates a minimal valid response matching the output_type schema. For VerdictModel this means it will produce a dict with all required fields. If the test needs control over the response, use `TestModel(custom_result_args={"decision": "approve", "confidence": 0.9, "rationale": "test", "conditions": []})`.
