Create `infra/entrypoint.sh`:
```bash
#!/bin/sh
set -e

# Wait for Postgres to be ready (belt-and-suspenders — compose health check covers this,
# but entrypoint adds application-level wait for migration safety)
until pg_isready -h db -U eclusa -d eclusa 2>/dev/null; do
  echo "Waiting for db..."
  sleep 1
done

echo "Running alembic migrations..."
uv run alembic upgrade head

echo "Starting web API..."
exec uv run uvicorn adapters.web.app:create_app --factory --host 0.0.0.0 --port 8000 --workers 1
```

Make it executable (add `chmod +x` note in the file; the Dockerfile will RUN chmod).

Overwrite `docker-compose.yml` with all 5 services:
```yaml
services:
  db:
    image: paradedb/paradedb:latest
    environment:
      POSTGRES_USER: eclusa
      POSTGRES_PASSWORD: eclusa
      POSTGRES_DB: eclusa
    ports:
      - "5432:5432"
    volumes:
      - pg_data:/var/lib/postgresql
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U eclusa -d eclusa"]
      interval: 5s
      timeout: 5s
      retries: 10
      start_period: 10s

  proxy:
    image: mitmproxy/mitmproxy:latest
    command: mitmdump -s /app/proxy/addon.py --listen-port 8080 --listen-host 0.0.0.0 --set block_global=false
    volumes:
      - .:/app
    ports:
      - "8080:8080"
    depends_on:
      db:
        condition: service_healthy
    environment:
      DATABASE_URL: postgresql://eclusa:eclusa@db:5432/eclusa

  web:
    build:
      context: .
      dockerfile: Dockerfile
    command: ["/app/infra/entrypoint.sh"]
    ports:
      - "8000:8000"
    depends_on:
      db:
        condition: service_healthy
    environment:
      DATABASE_URL: postgresql://eclusa:eclusa@db:5432/eclusa
      JWT_SECRET: "${JWT_SECRET:-dev-secret-change-me}"
      HTTP_PROXY: "http://proxy:8080"
      HTTPS_PROXY: "http://proxy:8080"
    healthcheck:
      test: ["CMD-SHELL", "curl -sf http://localhost:8000/api/health || exit 1"]
      interval: 10s
      timeout: 5s
      retries: 10
      start_period: 30s

  executor:
    build:
      context: .
      dockerfile: Dockerfile
    command: uv run python -m executor.loop
    depends_on:
      db:
        condition: service_healthy
    environment:
      DATABASE_URL: postgresql://eclusa:eclusa@db:5432/eclusa
      HTTP_PROXY: "http://proxy:8080"
      HTTPS_PROXY: "http://proxy:8080"
    restart: unless-stopped

  ui:
    build:
      context: ./ui
      dockerfile: Dockerfile
    ports:
      - "80:80"
    depends_on:
      web:
        condition: service_healthy

volumes:
  pg_data:
```

The `web` and `executor` services need a Python Dockerfile at the project root. Create `Dockerfile` (root):
```dockerfile
FROM python:3.12-slim

# Install system deps
RUN apt-get update && apt-get install -y curl postgresql-client && rm -rf /var/lib/apt/lists/*

# Install uv
COPY --from=ghcr.io/astral-sh/uv:latest /uv /usr/local/bin/uv

WORKDIR /app
COPY pyproject.toml uv.lock ./
RUN uv sync --frozen --no-dev

COPY . .
RUN chmod +x infra/entrypoint.sh

ENV PATH="/app/.venv/bin:$PATH"
ENV PYTHONPATH="/app"
```
