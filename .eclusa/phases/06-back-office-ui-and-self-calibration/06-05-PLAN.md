Create `ui/src/ws/useSessionStream.ts`:
```typescript
import useWebSocket, { ReadyState } from 'react-use-websocket'
import { useState, useEffect } from 'react'
import { useAuth } from '@/auth/AuthProvider'

export function useSessionStream(sessionId: string | null) {
  const { token } = useAuth()
  const [messages, setMessages] = useState<SessionMessage[]>([])
  const wsUrl = sessionId && token
    ? `ws://localhost:8000/ws/sessions/${sessionId}?token=${token}`
    : null

  const { lastJsonMessage, readyState } = useWebSocket(wsUrl, {
    shouldReconnect: () => true,
    reconnectAttempts: 10,
    reconnectInterval: 3000,
    share: true,
  })

  useEffect(() => {
    if (lastJsonMessage) {
      const msg = lastJsonMessage as SessionMessage
      if (msg.role) setMessages(prev => [...prev, msg])  // live messages
      if (Array.isArray(lastJsonMessage)) setMessages(lastJsonMessage)  // initial batch
    }
  }, [lastJsonMessage])

  return { messages, isConnected: readyState === ReadyState.OPEN }
}
```
