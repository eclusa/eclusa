```typescript
import { useQuery } from '@tanstack/react-query'
import { useAuth } from '@/auth/AuthProvider'
import type { PaginationState } from '@tanstack/react-table'

export function useLedgerAsOf(asOf: string, pagination: PaginationState) {
  const { getAuthHeader } = useAuth()
  const params = new URLSearchParams({
    as_of: asOf,
    page: String(pagination.pageIndex),
    size: String(pagination.pageSize),
  })
  return useQuery({
    queryKey: ['ledger', asOf, pagination.pageIndex, pagination.pageSize],
    queryFn: () => fetch(`/api/ledger?${params}`, { headers: getAuthHeader() }).then(r => r.json()),
    keepPreviousData: true,
  })
}
```

```tsx
// Per D-22: show bi-temporal timestamps compactly
// t_valid: "valid since YYYY-MM-DD" (or with end if t_invalid set)
// t_created: "created YYYY-MM-DD" + expired if t_expired set
function BiTemporalBadge({ fact }) {
  const validUntil = fact.t_invalid ? formatDate(fact.t_invalid) : 'current'
  const expiredAt = fact.t_expired ? formatDate(fact.t_expired) : 'active'
  return (
    <div className="flex gap-2 text-xs font-mono text-zinc-500">
      <span>valid: {formatDate(fact.t_valid)} → {validUntil}</span>
      <span>·</span>
      <span>record: {formatDate(fact.t_created)} → {expiredAt}</span>
    </div>
  )
}
```
