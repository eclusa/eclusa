Create `ui/src/api/metrics.ts`:
```typescript
import { useQuery } from '@tanstack/react-query'
import { useAuth } from '@/auth/AuthProvider'

export interface CalibrationMetrics {
  gate_necessity: number | null
  orchestrator_absorption: number | null
  resolution_latency: number | null
  decision_durability: number | null
  cascade_rework: number | null
  model_convergence: number | null
  minority_accuracy: number | null
  fanout_necessity: number | null
}

export function useCalibrationMetrics() {
  const { getAuthHeader } = useAuth()
  return useQuery<CalibrationMetrics>({
    queryKey: ['metrics', 'calibration'],
    queryFn: () => fetch('/api/metrics', { headers: getAuthHeader() }).then(r => r.json()),
    refetchInterval: 60_000,
    staleTime: 30_000,
  })
}
```

```tsx
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'

export function EmptyStateGuard({ value, title, children }: { value: number | null | undefined; title: string; children: React.ReactNode }) {
  if (value === null || value === undefined) {
    return (
      <Card className="bg-zinc-900/50 border-zinc-800 opacity-60">
        <CardHeader className="pb-2"><CardTitle className="text-xs font-mono text-zinc-500 uppercase tracking-widest">{title}</CardTitle></CardHeader>
        <CardContent><p className="text-sm text-zinc-600">Requires 10+ resolved gates to compute.</p></CardContent>
      </Card>
    )
  }
  return <>{children}</>
}
```

```tsx
import { useCalibrationMetrics, CalibrationMetrics } from '@/api/metrics'
import { MetricCard } from '@/components/metrics/MetricCard'
import { EmptyStateGuard } from '@/components/metrics/EmptyStateGuard'
import { MetricChart } from '@/components/metrics/MetricChart'
import { Skeleton } from '@/components/ui/skeleton'

const METRIC_DEFINITIONS = [ /* ... as defined in Task 1 */ ]

export default function MetricsPage() {
  const { data, isLoading, dataUpdatedAt } = useCalibrationMetrics()

  return (
    <div className="p-6 space-y-6">
      <div className="flex items-center justify-between">
        <h1 className="text-xs font-mono text-zinc-400 uppercase tracking-widest">Self-Calibration Metrics</h1>
        {dataUpdatedAt > 0 && (
          <span className="text-xs text-zinc-600 font-mono">
            updated {new Date(dataUpdatedAt).toLocaleTimeString()}
          </span>
        )}
      </div>

      {isLoading ? (
        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-4 gap-4">
          {Array.from({ length: 8 }).map((_, i) => <Skeleton key={i} className="h-32 bg-zinc-800/50" />)}
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-4 gap-4">
          {METRIC_DEFINITIONS.map(def => {
            const value = data ? data[def.key as keyof CalibrationMetrics] : null
            return (
              <EmptyStateGuard key={def.key} value={value} title={def.title}>
                <MetricCard title={def.title}>
                  <MetricChart
                    value={value as number}
                    type={def.type as 'rate' | 'latency'}
                    title={def.title}
                    description={def.description}
                  />
                </MetricCard>
              </EmptyStateGuard>
            )
          })}
        </div>
      )}
    </div>
  )
}
```
