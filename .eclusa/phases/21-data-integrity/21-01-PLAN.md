```python
async def resolve_gate(conn: asyncpg.Connection, stage_id: str, actor_id: str) -> None:
    async with conn.transaction():
        # Fetch cascade_id for scoped NOTIFY (D-05)
        row = await conn.fetchrow(
            "SELECT cascade_id FROM stage WHERE id = $1::uuid", stage_id
        )
        if row is None:
            raise ValueError(f"Stage not found: {stage_id}")
        cascade_id = str(row["cascade_id"])

        # INTEG-01: atomic guard — only resolves if currently 'blocked'
        # RETURNING id returns a row only if the UPDATE fired; empty means already resolved
        updated = await conn.fetchrow(
            """
            UPDATE stage
            SET state = 'resolved', resolved_at = NOW(), resolved_by = $1::uuid
            WHERE id = $2::uuid AND type = 'gate' AND state = 'blocked'
            RETURNING id
            """,
            actor_id,
            stage_id,
        )
        if updated is None:
            raise GateAlreadyResolvedError(
                f"Gate {stage_id} is not in 'blocked' state — already resolved or wrong type"
            )

        await conn.execute(
            """
            INSERT INTO ledger_entry (id, stage_id, cascade_id, actor_id, type, content, schema_version)
            SELECT gen_random_uuid(), s.id, s.cascade_id, $1::uuid,
                   'gate_resolved',
                   jsonb_build_object('stage_id', s.id::text),
                   $2
            FROM stage s WHERE s.id = $3::uuid
            """,
            actor_id,
            SYSTEM_SCHEMA_VERSION,
            stage_id,
        )

        payload = json.dumps({"stage_id": stage_id, "cascade_id": cascade_id})
        await conn.execute("SELECT pg_notify($1, $2)", "stage_changed", payload)
        await conn.execute(
            "SELECT pg_notify($1, $2)", f"stage_changed:{cascade_id}", payload
        )
```

In db/queries/trace_chain.sql, fix the recursive case (INTEG-02).

The bug: line 29 writes `c.intent_id AS cascade_id` which overwrites the cascade_id column with an intent_id on every recursive hop. The recursive case was intended to walk sub-cascades, but cascades do not nest via intent_id — an artifact belongs to exactly one cascade and that does not change as the CTE recurses.

The correct fix: remove the recursive UNION ALL entirely. The base case already captures the full ancestry (artifact → session → stage → cascade → intent) in a single row. The recursive case adds no new information and corrupts cascade_id.

Replace the entire file with:

```sql
-- Trace chain: artifact → session → stage → cascade → intent
-- Returns the full ancestry of an artifact in a single query.
-- $1 = artifact UUID to trace
--
-- INTEG-02 fix: removed the recursive UNION ALL case.
-- The base case captures all five ancestry columns in one row.
-- The prior recursive case overwrote cascade_id with intent_id on every hop — incorrect.
-- Cascades do not nest via intent_id; a single join is sufficient.
-- Source: eclusa.md §3.6; CONTEXT.md D-17; ARCHITECTURE.md Pattern 3

SELECT
  a.id           AS artifact_id,
  a.session_id,
  a.stage_id,
  a.cascade_id,
  a.intent_id,
  -- Enriched context from joined tables
  s.type         AS stage_type,
  s.state        AS stage_state,
  c.state        AS cascade_state,
  i.raw          AS intent_raw,
  i.source       AS intent_source
FROM artifact a
LEFT JOIN stage   s ON s.id = a.stage_id
LEFT JOIN cascade c ON c.id = a.cascade_id
LEFT JOIN intent  i ON i.id = a.intent_id
WHERE a.id = $1
LIMIT 1;
```
