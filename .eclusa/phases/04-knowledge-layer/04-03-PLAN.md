class IREntity(IRElement):
    name: str
    description: str | None = None
    type: Literal["model", "message", "type", "table", "interface"] = "model"

class IRField(IRElement):
    entity_name: str
    field_name: str
    field_type: str
    nullable: bool = True
    is_primary_key: bool = False
    is_foreign_key: bool = False
    foreign_ref: str | None = None

class IRRelation(IRElement):
    from_entity: str
    to_entity: str
    relation_type: Literal["one_to_one", "one_to_many", "many_to_many", "unknown"]

class IROperation(IRElement):
    name: str
    method: str | None = None
    path: str | None = None

class SchemaIR(BaseModel):
    source_format: Literal["openapi", "prisma", "sql_ddl", "graphql_sdl", "protobuf"]
    entities: list[IREntity] = []
    fields: list[IRField] = []
    relations: list[IRRelation] = []
    operations: list[IROperation] = []
    constraints: list[IRConstraint] = []
    warnings: list[ParseWarning] = []
```

    Create schema_commons/parsers/prisma.py as a ~150-line hand-rolled tokenizer. No external library needed — Prisma's DSL is a simple block-structured format.

    Algorithm:
    1. Strip line comments (// ...) from each line
    2. Scan line by line for `model ModelName {` — start of model block
    3. Inside a model block, collect lines until closing `}` at the start of a line
    4. For each non-empty field line inside a model block:
       - First token = field_name
       - Second token = field_type (may end in `?` for nullable, `[]` for array)
       - Remaining tokens = attributes (e.g., @id, @unique, @relation, @default)
    5. Extract @relation target from `@relation(fields: [...], references: [...])` — the model type name IS the relation target
    6. Produce IREntity per model, IRField per field, IRRelation per @relation field

    Key edge cases:
    - `id String @id` — is_primary_key=True, nullable=False (String without ? is non-nullable in Prisma)
    - `posts Post[] @relation(...)` — creates IRRelation(from_entity="User", to_entity="Post", relation_type="one_to_many")
    - `userId String` (FK field without @relation) — just a regular field
    - Multi-line `@relation(...)` with newlines inside — use a simple state machine to handle

    Type normalization: strip `?` and `[]` to get base_type; set nullable from presence of `?`; is_primary_key from `@id`.

    from proto_schema_parser.parser import Parser as ProtoParser
    from proto_schema_parser.ast import Message, Field, Service, RPC, File
    from schema_commons.ir import SchemaIR, IREntity, IRField, IROperation, ParseWarning

    def _field_type_str(field) -> str:
        """Extract type string from proto field, including repeated prefix."""
        cardinality = getattr(field, "cardinality", None)
        type_name = getattr(field, "type", str(field))
        if hasattr(type_name, "value"):
            type_name = type_name.value
        if cardinality and str(cardinality) in ("FieldCardinality.REPEATED", "repeated"):
            return f"repeated {type_name}"
        return str(type_name)

    def parse_protobuf(text: str) -> SchemaIR:
        ir = SchemaIR(source_format="protobuf")
        try:
            proto_file = ProtoParser().parse(text)
        except Exception as exc:
            ir.warnings.append(ParseWarning(location="root", message=str(exc)))
            return ir
        _walk_elements(proto_file.file_elements, ir, prefix="")
        return ir

    def _walk_elements(elements, ir: SchemaIR, prefix: str) -> None:
        for element in elements:
            if isinstance(element, Message):
                entity_ref = f"{prefix}message.{element.name}" if not prefix else f"{prefix}.{element.name}"
                ir.entities.append(IREntity(
                    source_ref=entity_ref,
                    name=element.name,
                    type="message",
                ))
                for body_element in element.elements:
                    if isinstance(body_element, Field):
                        ir.fields.append(IRField(
                            source_ref=f"{entity_ref}.field.{body_element.name}",
                            entity_name=element.name,
                            field_name=body_element.name,
                            field_type=_field_type_str(body_element),
                            nullable=True,  # proto3 fields are technically always optional
                            is_primary_key=False,
                        ))
            elif isinstance(element, Service):
                for rpc_element in element.elements:
                    if isinstance(rpc_element, RPC):
                        ir.operations.append(IROperation(
                            source_ref=f"service.{element.name}.rpc.{rpc_element.name}",
                            name=rpc_element.name,
                            method="rpc",
                            path=None,
                        ))
    ```

    Note: proto-schema-parser's AST field names may vary slightly between versions. Read the actual installed version's source or use `dir()` on the AST nodes to confirm field names. Adjust the implementation if `Field`, `Message`, `Service`, `RPC` attribute names differ from what's shown above. The core logic (walk messages for entities, walk services for operations) is stable.
