Implement the first half of the temporal knowledge graph: episode ingestion (KG-01) and entity extraction + resolution (KG-02). These are the entry point to the KG pipeline — every piece of knowledge flows through episode → entity → fact.

Purpose: Episodes preserve raw data exactly as received (KG-01). Entity extraction via LLM produces structured entities from episode content (D-13). From RESEARCH.md Pattern 7 (entity resolution):
```python
RESOLUTION_THRESHOLD = 0.88  # cosine similarity threshold

async def resolve_entity(name: str, embedding: list[float], conn) -> str | None:
    # Step 1: exact name match
    row = await conn.fetchrow("SELECT id FROM entity WHERE LOWER(name) = LOWER($1)", name)
    if row: return str(row["id"])
    # Step 2: embedding similarity >= threshold
    row = await conn.fetchrow(
        "SELECT id, 1 - (embedding <=> $1::vector) AS similarity "
        "FROM entity WHERE embedding IS NOT NULL "
        "AND 1 - (embedding <=> $1::vector) >= $2 "
        "ORDER BY embedding <=> $1::vector LIMIT 1",
        embedding, RESOLUTION_THRESHOLD,
    )
    return str(row["id"]) if row else None
```

    ```python
    Episodes preserve raw data exactly as received (KG-01, D-12).
    Every piece of knowledge flows through episode first.

    import json
    import logging
    from datetime import datetime, timezone
    import asyncpg

    logger = logging.getLogger(__name__)

    SYSTEM_SCHEMA_VERSION = "0001"


    async def ingest_episode(
        raw_data: dict,
        source: str,
        reference_ts: datetime,
        conn: asyncpg.Connection,
    ) -> str:
        """Write raw data to episode table. Returns episode UUID.

        raw_data is stored as-is in JSONB — no transformation (KG-01).
        reference_ts is the event-world time, not the ingestion time.
        """
        episode_id = await conn.fetchval(
            """
            INSERT INTO episode (id, source, raw_data, reference_ts, schema_version)
            VALUES (gen_random_uuid(), $1, $2::jsonb, $3, $4)
            RETURNING id::text
            """,
            source,
            json.dumps(raw_data),
            reference_ts,
            SYSTEM_SCHEMA_VERSION,
        )
        logger.debug("Ingested episode %s from source=%s", episode_id, source)
        return episode_id
    ```

    Create knowledge/extract.py following RESEARCH.md Pattern 6, then add entity persistence wiring:
    ```python
    from pydantic import BaseModel
    from pydantic_ai import Agent
    import asyncpg
    from knowledge.resolve import resolve_entity

    class ExtractedEntity(BaseModel):
        name: str
        type: str  # "person", "project", "api", "decision", "rule", "concept"
        summary: str

    class EntityList(BaseModel):
        entities: list[ExtractedEntity]

    _agent = Agent("anthropic:claude-haiku-4-5", output_type=EntityList)

    async def extract_entities_from_episode(
        raw_text: str,
        conn: asyncpg.Connection,
    ) -> list[ExtractedEntity]:
        """Extract entities from episode text and persist new entities to entity table.

        For each ExtractedEntity:
        - Calls resolve_entity() to check for existing match (name-exact or cosine)
        - If None: INSERTs a new row into entity table (KG-02 durable persistence)
        - If existing id: skips INSERT (deduplication)

        Embedding is not set here — the embedding pipeline (04-04) handles that separately.
        """
        result = await _agent.run(
            f"Extract all named entities (people, projects, APIs, business rules, "
            f"decisions) from the following content. Be concise in summaries.\n\n{raw_text}"
        )
        entities = result.output.entities

        # Persist new entities (KG-02: durable entity persistence)
        for entity in entities:
            existing_id = await resolve_entity(entity.name, [], conn)
            if existing_id is None:
                await conn.execute(
                    """
                    INSERT INTO entity (id, name, type, summary, created_at, updated_at)
                    VALUES (gen_random_uuid(), $1, $2, $3, NOW(), NOW())
                    """,
                    entity.name, entity.type, entity.summary,
                )

        return entities
    ```
