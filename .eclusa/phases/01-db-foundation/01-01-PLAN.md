```yaml
services:
  db:
    image: paradedb/paradedb:latest
    environment:
      POSTGRES_USER: eclusa
      POSTGRES_PASSWORD: eclusa
      POSTGRES_DB: eclusa
    ports:
      - "5432:5432"
    volumes:
      - pg_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U eclusa -d eclusa"]
      interval: 5s
      timeout: 5s
      retries: 10
      start_period: 10s

volumes:
  pg_data:
```

Note: `paradedb/paradedb:latest` ships with both pgvector and pg_search pre-compiled against the bundled Postgres version. No additional extension installation steps are needed in an init script — they are available via `CREATE EXTENSION IF NOT EXISTS vector` and `CREATE EXTENSION IF NOT EXISTS pg_search` in the migration.

Create `tests/conftest.py` with a session-scoped fixture that:
1. Starts a `PostgresContainer("paradedb/paradedb:latest")` — same image as docker-compose (CRITICAL: not postgres:17)
2. Runs Alembic migrations programmatically using `alembic.config.Config` + `alembic.command.upgrade`
3. Yields a connection URL string for use in test-scoped connection fixtures
4. Provides an `asyncpg` connection fixture for integration tests

```python
import asyncio
import pytest
import asyncpg
from alembic.config import Config
from alembic.command import upgrade
from testcontainers.postgres import PostgresContainer


@pytest.fixture(scope="session")
def pg_container():
    """Real PG17 instance with pgvector + pg_search extensions (paradedb image)."""
    with PostgresContainer("paradedb/paradedb:latest") as pg:
        yield pg


@pytest.fixture(scope="session")
def db_url(pg_container) -> str:
    """Sync connection URL for Alembic."""
    # testcontainers returns a sync URL; convert for asyncpg
    return pg_container.get_connection_url()


@pytest.fixture(scope="session", autouse=True)
def apply_migrations(pg_container, db_url):
    """Run all Alembic migrations against the test container."""
    alembic_cfg = Config("alembic.ini")
    # Override the URL from alembic.ini with the test container's URL
    # Convert to async driver URL for env.py
    async_url = db_url.replace("postgresql://", "postgresql+asyncpg://", 1)
    alembic_cfg.set_main_option("sqlalchemy.url", async_url)
    upgrade(alembic_cfg, "head")


@pytest.fixture
async def conn(pg_container) -> asyncpg.Connection:
    """Per-test asyncpg connection to the test database."""
    # Use the raw asyncpg DSN (not SQLAlchemy URL)
    url = pg_container.get_connection_url()
    # Convert sqlalchemy format: postgresql://user:pass@host:port/db
    conn = await asyncpg.connect(url.replace("postgresql+asyncpg://", "postgresql://", 1)
                                     .replace("postgresql://", "postgresql://", 1))
    yield conn
    await conn.close()
```

Note: `apply_migrations` is `autouse=True` and `scope="session"` so migrations run once before any test accesses the DB.
